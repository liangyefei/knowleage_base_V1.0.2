## 2. 设计方案

### 2.1 总体策略

采用 **"自动检测 + 显式覆盖 + 集中式模板解析"** 三层设计：

1. **自动检测**：用 `langdetect` 从用户 query 检测语言，缓存结果（同 query 二次请求不重复检测）。
2. **显式覆盖**：所有问答端点新增 `language` 参数（`auto`/`zh`/`en`，默认 `auto`），前端可强制指定。
3. **集中式解析**：新增 `resolve_prompt_name(category, prompt_name, language)` 统一解析，命中英文模板则切换，未命中则优雅降级回中文模板（不报错）。
4. **特性开关/灰度回滚**：新增环境变量 `HAISI_LANG_AUTO_SWITCH`（默认 `"1"` 启用，命名与 QPS 优化文档的 `HAISI_*` env 约定一致）。设为 `"0"` 时 `resolve_language` 短路返回 `zh`，所有端点强制中文（=今天行为，零变更），作为上线后如出现英文误判时的秒级 kill-switch，且无需重启；并对接现有灰度发布基建按比例放量。

> 不在 LLM 层做"请用英文回答"的指令注入——而是通过切换整段 system prompt 实现语言控制，与现有 `en_*` 模板设计一致，且不污染 Dashscope 前缀缓存（system 仍是稳定前缀）。

### 2.2 语言检测模块

**新增文件**：`server/chat/pipeline/language_detect.py`

```python
"""语言检测：基于 langdetect，带 LRU 缓存、CJK 快速通道与代码主导识别。"""
import os
import re
from functools import lru_cache
from typing import Literal

from configs.logger_manager import logger


def _auto_switch_enabled() -> bool:
    """kill-switch：HAISI_LANG_AUTO_SWITCH 默认启用；设为 "0"/"false" 时关闭，
    resolve_language 短路返回 zh（=今天行为，零变更、无需重启）。"""
    return os.environ.get("HAISI_LANG_AUTO_SWITCH", "1") not in ("0", "false", "False", "")

# langdetect 默认带随机性，固定 seed 保证可复现
try:
    from langdetect import detect_langs, DetectorFactory
    DetectorFactory.seed = 0
    _HAS_LANGDETECT = True
except ImportError:  # 依赖缺失时 fail-open，按中文处理
    _HAS_LANGDETECT = False

Lang = Literal["zh", "en"]

# ASCII 字母占比阈值：英文短句（如 "hi"）langdetect 不稳，用字符统计兜底
_EN_ASCII_RATIO = 0.6
_MIN_LEN_FOR_DETECT = 3  # 短于此长度走字符统计快速通道

# 代码符号集（散文中罕见、代码中密集）：用于识别"纯代码/代码主导"输入
_CODE_SYMBOLS = set("{}()[];=<>+-*/\\|&!@#$%^~:`")
# 代码符号占 (符号+字母) 比例阈值：≥ 此值判定为代码主导
_CODE_SYMBOL_RATIO = 0.15
# 代码关键词 + 紧邻代码标点：要求关键词像代码那样被使用（def foo( / return x;）
# 单独出现在散文里的 "return"/"class" 不算（避免 "How to use return" 误判）
_CODE_KW_WITH_PUNCT = re.compile(
    r'\b(?:def|class|import|from|return|public|private|protected|void|'
    r'int|float|double|char|bool|var|let|const|async|await|new|try|catch|'
    r'finally|throw|lambda|fn|func|struct|enum|interface|extends|implements|'
    r'package|namespace|using|#include)\b[^a-zA-Z]{0,3}[(:;{=]'
)
# markdown 代码块 / 行内代码，检测前剥离，用剩余自然语言片段判定语言
_FENCED_CODE = re.compile(r"```[a-zA-Z]*\n?.*?\n```", re.DOTALL)
_INLINE_CODE = re.compile(r"`[^`\n]+`")


def _strip_code_blocks(q: str) -> str:
    """剥离 markdown 代码块（```...```）与行内代码（`...`），返回自然语言片段。

    用于"英文 prose + 嵌入代码块"场景：剥离代码后对 prose 检测，
    避免 prose 被代码符号拉高密度而误判为"代码主导"。
    """
    q = _FENCED_CODE.sub(" ", q)
    q = _INLINE_CODE.sub(" ", q)
    return q.strip()


def _is_code_dominant(q: str) -> bool:
    """判断输入是否为代码主导（缺少可识别的自然语言信号）。

    代码主导时不切英文模板——保留默认中文，避免：
      - 纯代码被 ASCII 占比误判为英文（全 ASCII 字符）
      - langdetect 把 C/Python 代码里的 int/return/def 等英文词当成英文自然语言

    判据（满足任一）：
      1. 代码符号密度 ≥ _CODE_SYMBOL_RATIO（{}()[];=<> 等占比高）
      2. 代码关键词以代码方式使用（紧邻 ( : ; { = 等标点）
    """
    if not q:
        return False
    code_syms = sum(1 for c in q if c in _CODE_SYMBOLS)
    letters = sum(1 for c in q if c.isalpha())
    total = code_syms + letters
    if total > 0 and code_syms / total >= _CODE_SYMBOL_RATIO:
        return True
    if _CODE_KW_WITH_PUNCT.search(q):
        return True
    return False


def _is_cjk_dominant(q: str) -> bool:
    """CJK 字符数 ≥ ASCII 字母数 → 判为中文主导，直接返回 zh 绕过 langdetect。

    中文主流量走此快速通道（亚 0.1ms、不阻塞 async 事件循环），langdetect 仅
    用于 ASCII 占优的英文候选流量。用"占优"而非"出现"，避免
    'What is 海思's advantage?'（英文 prose + 单个中文词）被误判 zh。
    """
    cjk = sum(1 for c in q if "\u4e00" <= c <= "\u9fff")
    ascii_letters = sum(1 for c in q if c.isalpha() and ord(c) < 128)
    return cjk > 0 and cjk >= ascii_letters


def detect_language(query: str) -> Lang:
    """检测 query 主语言，返回 'zh' 或 'en'。

    优先级：
    1. 空/纯标点 → 'zh'（默认）
    2. 剥离 markdown 代码块/行内代码后，剩余自然语言为空 → 纯代码 → 'zh'
    3. CJK 占优（中文主流量）→ 'zh'，绕过 langdetect（不阻塞 async 事件循环）
    4. 剩余自然语言仍"代码主导"（裸代码未用代码块包裹）→ 'zh'
    5. 极短文本（≤3 字符）→ ASCII 字母占比判定（"hi"/"ok"/"你好"）
    6. langdetect 检测，top1 命中 zh*/en* 即返回（剥离代码后 prose 主导信号）
    7. langdetect 异常/非中英 → 按字符占比兜底
    8. 全部失败 → 'zh'（fail-open，保持现有中文体验）

    关键：代码块先剥离再检测，使"英文提问 + 内嵌代码块"正确判为 en，
    而纯代码（无自然语言）保留默认 zh，不被误判为英文提问；CJK 占优快速通道
    让中文主流量绕过 langdetect，避免 async 端点阻塞事件循环。
    """
    if not query or not query.strip():
        return "zh"
    q = query.strip()

    # 剥离代码块，用剩余自然语言片段判定
    prose = _strip_code_blocks(q)
    if not prose:
        # 纯代码（无任何自然语言片段）→ 保留默认中文
        return "zh"

    # CJK 占优快速通道：中文主流量直接 zh，绕过 langdetect（不阻塞 async 事件循环）
    if _is_cjk_dominant(prose):
        return "zh"

    # 代码主导（裸代码未用代码块包裹，符号密度高/含代码关键词）→ 保留中文
    if _is_code_dominant(prose):
        logger.debug(f"[lang] code-dominant input, keep default zh | query={q[:40]}")
        return "zh"

    # 极短文本：langdetect 不稳，用字符占比
    if len(prose) <= _MIN_LEN_FOR_DETECT or not _HAS_LANGDETECT:
        return _by_ascii_ratio(prose)

    try:
        langs = detect_langs(prose)
        if langs:
            top = langs[0].lang  # 如 'en', 'zh-cn', 'zh-tw'
            if top.startswith("en"):
                return "en"
            if top.startswith("zh"):
                return "zh"
        # langdetect 给出非中英文（如 ja/ko/fr）→ 按字符占比判定
        return _by_ascii_ratio(prose)
    except Exception as e:
        logger.debug(f"[lang] detect 失败 fail-open=zh: {e} | query={q[:40]}")
        return _by_ascii_ratio(prose)


def _by_ascii_ratio(q: str) -> Lang:
    """按 ASCII 字母占比判定：≥_EN_ASCII_RATIO 判英文，否则中文。"""
    letters = [c for c in q if c.isalpha()]
    if not letters:
        return "zh"
    ascii_letters = [c for c in letters if ord(c) < 128]
    return "en" if len(ascii_letters) / len(letters) >= _EN_ASCII_RATIO else "zh"


@lru_cache(maxsize=2048)
def detect_language_cached(query: str) -> Lang:
    """带 LRU 缓存的检测：同一 query 二次请求 O(1) 命中。

    query 作为 key，缓存命中率取决于用户重复提问频率；langdetect 单次 ~0.1ms，
    缓存主要意义是避免高频重复问题重复检测 + 保证同请求内多次调用结果一致。
    """
    return detect_language(query)


def resolve_language(query: str, explicit: str = "auto") -> Lang:
    """融合显式参数与自动检测：explicit != 'auto' 时直接采用，否则检测。

    kill-switch 优先：HAISI_LANG_AUTO_SWITCH=0 时直接返回 zh，忽略显式覆盖，
    作为上线后英文误判的秒级回滚开关（=今天行为，零变更）。
    """
    if not _auto_switch_enabled():
        return "zh"
    if explicit == "en":
        return "en"
    if explicit == "zh":
        return "zh"
    return detect_language_cached(query)
```

> **设计要点**：
> - `DetectorFactory.seed = 0` 保证可复现（langdetect 默认随机性会导致同输入不同结果）。
> - 极短文本（`≤3` 字符）和问候语走字符占比快速通道，避免 langdetect 对 "hi"/"ok" 这类短词误判。
> - **代码主导识别**（关键）：`langdetect` 对纯代码要么抛异常、要么把 `int/return/def` 等英文词误判为英文自然语言 → 会把纯代码输入误判为 `en` 切英文模板。修复：先剥离 markdown 代码块/行内代码，对剩余自然语言片段检测；剩余仍为空（纯代码）或符号密度高/含代码关键词（裸代码）→ 判为"代码主导"，保留默认 `zh`，不切英文模板。这样"英文提问 + 内嵌代码块"仍正确判为 `en`，而纯代码不会误切英文。
> - 全链路 fail-open 到 `zh`，依赖缺失/检测异常都不影响现有中文体验。
> - `lru_cache` 保证同请求内多次调用（路由/生成/记忆抽取都可能调）结果一致且零重复开销。

### 2.3 模板名解析器

```python
def resolve_prompt_name(category: str, prompt_name: str, language: str) -> str:
    """根据语言解析最终模板名：language=='en' 时优先取 'en_{prompt_name}'，
    不存在则降级回 prompt_name（优雅降级，不报错）。

    用于 multi_knowledge_base_chat / file_chat / chat 等端点的模板选择层，
    把"语言 → 模板名"映射收敛到一处，避免各端点各自 if-else。

    Args:
        category: 模板类别（如 "knowledge_base_chat"）
        prompt_name: 请求传入的基础模板名（如 "default"/"text"/"think"/"empty"）
        language: "zh" 或 "en"

    Returns:
        最终模板名。en 命中则 "en_{prompt_name}"，否则 prompt_name。
    """
    if language != "en":
        return prompt_name
    from configs import prompt_loader
    templates = prompt_loader.get_loader().load()
    cat = templates.get(category, {})
    en_name = f"en_{prompt_name}"  # chitchat → en_chitchat，命名统一，无需特判
    if isinstance(cat.get(en_name), dict):
        return en_name
    return prompt_name  # 降级回中文模板
```

```python
import pytest
from server.chat.pipeline.language_detect import detect_language, resolve_language

class TestDetectLanguage:
    @pytest.mark.parametrize("query,expected", [
        ("Hi3798MV310 的启动流程是什么？", "zh"),
        ("What is the boot process of Hi3798MV310?", "en"),
        ("hi", "en"),                       # 极短英文走字符占比
        ("你好", "zh"),                      # 极短中文走字符占比
        ("GPIO 配置", "zh"),                  # 含英文缩写但中文为主
        ("How to configure GPIO?", "en"),
        ("", "zh"),                          # 空串 fail-open
        ("12345", "zh"),                     # 纯数字 fail-open
        ("OK", "en"),                        # 问候词
        ("Thanks a lot", "en"),
        # ===== 代码主导识别（核心）=====
        ("def foo():\n    return 42", "zh"),          # 纯 Python 代码 → 不切英文
        ("int main() { return 0; }", "zh"),            # 纯 C 代码 → 不切英文
        ("GPIO_CFG = 0x1234;", "zh"),                  # 代码片段 → 不切英文
        ("x=1", "zh"),                                # 极短代码（符号密度高）→ 不切英文
        ("How to use return statement?", "en"),        # 散文含 code kw 但非代码主导
        ("why error?\n```python\ndef foo():\n    return 1\n```", "en"),  # 英文 prose + 代码块
        ("这段代码报错了\n```python\ndef foo(): pass\n```", "zh"),        # 中文 prose + 代码块
    ])
    def test_detect(self, query, expected):
        assert detect_language(query) == expected

    def test_resolve_explicit_override(self):
        assert resolve_language("你好", "en") == "en"
        assert resolve_language("hello", "zh") == "zh"
        assert resolve_language("你好", "auto") == "zh"
        assert resolve_language("hello", "auto") == "en"

    def test_kill_switch(self, monkeypatch):
        # kill-switch 关闭 → 强制 zh，忽略显式 en（=今天行为，秒级回滚）
        monkeypatch.setenv("HAISI_LANG_AUTO_SWITCH", "0")
        assert resolve_language("hello", "en") == "zh"
        # kill-switch 开启 → 正常检测/显式覆盖
        monkeypatch.setenv("HAISI_LANG_AUTO_SWITCH", "1")
        assert resolve_language("hello", "auto") == "en"
```

## 5. 风险与缓解

| 风险 | 影响 | 缓解 |
|------|------|------|
| `langdetect` 对短 query（型号名 "Hi3516"）误判 | 英文模板误触发，答案变英文 | 极短文本（≤3 字符）走字符占比快速通道；型号名含数字+ASCII 会被判 en，但这类 query 多为"Hi3516 启动流程"含中文，主语言仍 zh |
| 英文模板 `en_default` 不存在时降级回中文 | 英文提问走中文模板，模型仍可能用中文答 | `resolve_prompt_name` 优雅降级不报错；模板已在 §2.4 补齐，降级仅在配置缺失时发生 |
| 路由 LLM（`route_chat`）system prompt 为中文 | 英文 query 路由 JSON 的 `reasoning` 字段可能为中文（辅助字段，不影响路由决策）；`route_reasoning` 状态时间线已由 P1 i18n 修复 | 一期可接受；二期补 `route_chat.en_system`（§2.5.5 末） |
| **检索语种 vs 回答语种错配** | 英文 query 仍从（多为中文的）KB 检索，`en_` 模板要求英文作答 → 模型跨语种综合，有幻觉 / 中文泄漏风险 | `en_` 模板显式要求英文作答，模型可跨语种综合；若质量不达标，二期评估检索前 query 翻译为中文再检索、或多语种 embedding 召回 |
| 记忆抽取（`memory_extraction`）中文 prompt 从英文对话漏抽 | 长期记忆缺失部分英文事实 | 低频后台任务，一期可接受；二期按需补 `memory_extraction.en_system` |
| 同一对话中用户中英文混用 | 单次按 query 语言切模板，对话历史语言不一致可能让模型困惑 | 当前按"当前 query 语言"判定，符合"用户用英文提问则英文回答"的需求；多轮语言切换属预期行为 |
| `langdetect` 依赖虽在 requirements 但镜像未装 | 运行时 ImportError | `language_detect.py` try-import + `_HAS_LANGDETECT` 标志，缺失时 fail-open 到 zh，不阻断服务 |
